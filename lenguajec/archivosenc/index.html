Gestión de Archivos con Explicaciones en el Código

⚙️ 1. Ejemplos Individuales Comentados

Cada fragmento incluye comentarios detallados sobre el uso del puntero FILE * y los modos de apertura.

Modo "w" (Escritura destructiva)

modo_escritura.cModo "w"
#include <stdio.h>

int main() {
    /* 
      ¿POR QUÉ USAR PUNTEROS (FILE *)?
      Los archivos están en el disco duro y el programa en la RAM. 
      'FILE *f' crea un puntero (una dirección de memoria) que actúa 
      como un "control remoto" o canal para conectar el programa 
      con el archivo físico del sistema operativo.
    */
    FILE *f = fopen("datos.txt", "w");

    /* Validamos siempre si el puntero es NULL (si falló la apertura) */
    if (f == NULL) {
        printf("Error al crear el archivo.\n");
        return 1;
    }

    /* Escribimos texto usando el puntero 'f' */
    fprintf(f, "Modo escritura: reemplaza todo el contenido anterior.\n");

    /* Cerramos el archivo para liberar el recurso en el disco */
    fclose(f);
    return 0;
}

Modo "r" (Lectura)

modo_lectura.cModo "r"
#include <stdio.h>

int main() {
    /* Declaramos el puntero para abrir en modo lectura ("r") */
    FILE *f = fopen("datos.txt", "r");

    /* Si el archivo no existe, fopen devuelve NULL obligatoriamente */
    if (f == NULL) {
        printf("El archivo no existe todavía.\n");
        return 1;
    }

    char linea[100];
    /* Leemos línea por línea hasta llegar al final (EOF - End Of File) */
    while (fgets(linea, sizeof(linea), f) != NULL) {
        printf("%s", linea);
    }

    fclose(f); /* Siempre cerrar el canal */
    return 0;
}

Modo "a" (Adición / Append)

modo_append.cModo "a"
#include <stdio.h>

int main() {
    /* El modo "a" posiciona el cursor al final sin borrar lo previo */
    FILE *f = fopen("datos.txt", "a");
    
    if (f == NULL) return 1;

    /* Se agrega texto nuevo respetando lo que ya estaba escrito */
    fprintf(f, "Modo append: esta línea se añade al final.\n");
    
    fclose(f);
    return 0;
}

📦 2. Ejemplo Integrado en ZinjaI (Comentado)

Este script completo unifica los tres modos en un solo flujo de trabajo.

programa_integrado.c (ZinjaI)Completo
#include <stdio.h>
#include <stdlib.h>

int main() {
    /* Puntero único reutilizable para gestionar el archivo */
    FILE *archivo;
    char buffer[120];

    /* -------------------------------------------------- */
    /* PASO 1: Escritura inicial con Modo "w"             */
    /* -------------------------------------------------- */
    archivo = fopen("bitacora.txt", "w");
    if (archivo == NULL) {
        printf("Error al abrir en modo 'w'.\n");
        return 1;
    }
    fprintf(archivo, "Registro 1: Sistema iniciado correctamente.\n");
    fclose(archivo); /* Cerramos para asegurar que se guarde en disco */

    /* -------------------------------------------------- */
    /* PASO 2: Añadir datos con Modo "a" (Sin borrar nada)*/
    /* -------------------------------------------------- */
    archivo = fopen("bitacora.txt", "a");
    if (archivo == NULL) { return 1; }
    fprintf(archivo, "Registro 2: Se adjunta nueva información.\n");
    fclose(archivo);

    /* -------------------------------------------------- */
    /* PASO 3: Leer todo el contenido con Modo "r"        */
    /* -------------------------------------------------- */
    archivo = fopen("bitacora.txt", "r");
    if (archivo == NULL) { return 1; }

    printf("--- CONTENIDO LEÍDO DEL ARCHIVO ---\n");
    /* El ciclo lee línea por línea utilizando el puntero 'archivo' */
    while (fgets(buffer, sizeof(buffer), archivo) != NULL) {
        printf("%s", buffer);
    }
    
    fclose(archivo); /* Cierre final obligatorio */
    return 0;
}
